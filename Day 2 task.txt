7. Login Attempt Controller 

import java.util.Scanner;

{

    { 
        Scanner S=new Scanner(System.in); 
        System.out.println("Enter the maximum login attempts:"); 
        int maxAttempts=S.nextInt(); 
        System.out.println("Enter the current failed login attempts:"); 
        int failedAttempts=S.nextInt(); 
        if(failedAttempts<maxAttempts) 
        { 
            System.out.println("Login successful"); 
        } 
        else 
        { 
            System.out.println("Login failed"); 
            failedAttempts++; 
        } 

            String status=(failedAttempts>=maxAttempts)?"Account locked":"login attempts remaining"; 
            System.out.println(status); 

    } 
}

12. Ride Fare Calculator 

import java.util.Scanner; 
public class Ridefare 
{ 
    public static void main(String[] args) 
    { 
        Scanner S=new Scanner(System.in); 
        System.out.println("Enter the distance travelled in km:"); 
        int distance=S.nextInt(); 
        System.out.println("Enter the Waiting time in minutes:"); 
        int waitingTime=S.nextInt(); 
        System.out.println("Enter the status of the member (ordinary/Premium):"); 
        String status=S.next(); 
        if(distance<0||waitingTime<0) 
        { 
            System.out.println("Invalid input. Please enter positive values."); 
        } 
        else 
        { 
            int fare=50+distance*12; 
            if(waitingTime>5) 
            { 
                fare=fare+(waitingTime-5)*2; 
            } 
            fare=(status.equals("Premium")&&fare>300)?(fare*10/100):fare; 
            System.out.println("The total fare is: " + fare); 
        } 



    } 
}

14. Secure Transaction Validator 

import java.util.Scanner; 
public class Transaction 
{ 
    public static void main(String[] args) 
    { 
        Scanner S=new Scanner(System.in); 
        System.out.println("Account balance:"); 
        int balance=S.nextInt(); 
        System.out.println("Enter the transaction amount:"); 
        int transaction=S.nextInt(); 
        System.out.println("Daily transaction limit:"); 
        int limit=S.nextInt();  
        String status=(balance>0)?"active":"inactive"; 
        if(status.equals("active") && transaction>0 && transaction<=balance&&transaction<=limit) 
        { 
            System.out.println("Transaction is valid"); 
            System.out.println("Remaining balance: "+(balance-transaction)); 
            System.out.println("Transaction Status: Success"); 
        } 

    else 
        { 
            System.out.println("Transaction is invalid"); 
            System.out.println("Transaction Status: Failed"); 
        } 
    } 
}
